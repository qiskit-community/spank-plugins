cmake_minimum_required(VERSION 3.15)
set(CMAKE_C_STANDARD 11)
project(test_spank_qrmi LANGUAGES C)
enable_testing()

set(PLUGIN_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../../spank_qrmi)
# qrmi.h is generated by the QRMI build; by default use the copy fetched by the plugin build.
set(QRMI_INCLUDE_DIR "${PLUGIN_DIR}/build/deps/src/QRMI" CACHE PATH "Directory containing qrmi.h")
if(NOT EXISTS "${QRMI_INCLUDE_DIR}/qrmi.h")
  message(FATAL_ERROR "qrmi.h not found in ${QRMI_INCLUDE_DIR}; build spank_qrmi first or set QRMI_INCLUDE_DIR")
endif()

find_path(SLURM_INCLUDE_DIR NAMES slurm/spank.h REQUIRED)

file(WRITE "${CMAKE_BINARY_DIR}/generated/spank_qrmi_version.h"
"#define SPANK_QRMI_VERSION \"test\"
#define SPANK_QRMI_GIT_HASH \"test\"
")

# The test includes spank_qrmi.c directly and provides fake Slurm and QRMI functions,
# so it links neither libslurm nor libqrmi.
add_executable(test_spank_qrmi src/test_spank_qrmi.c ${PLUGIN_DIR}/buf.c ${PLUGIN_DIR}/strbuf.c)
target_include_directories(test_spank_qrmi PRIVATE
  ${PLUGIN_DIR} ${QRMI_INCLUDE_DIR} ${SLURM_INCLUDE_DIR} ${CMAKE_BINARY_DIR}/generated)
file(READ "${QRMI_INCLUDE_DIR}/qrmi.h" QRMI_HEADER_CONTENTS)
if(QRMI_HEADER_CONTENTS MATCHES "qrmi_log_callback_set")
  target_compile_definitions(test_spank_qrmi PRIVATE QRMI_HAS_LOG_CALLBACK=1)
endif()
target_compile_options(test_spank_qrmi
  PRIVATE -Wall -Wextra -Werror -pedantic -Wconversion -Wwrite-strings -Wfloat-equal
)

set(TEST_CASES
  no_qpu_option
  task_step_is_skipped
  missing_config_argument
  config_load_failure
  single_resource
  resource_list
  unknown_resource
  unknown_resource_in_list
  inaccessible_resource
  acquire_failure
  plugin_env_arguments
  plugin_env_argument_without_equals
  job_env_is_copied
  srun_debug_sets_rust_log
  job_env_entry_without_equals
  token_freed_with_matching_deallocator
  fresh_token_replaces_job_env_token
)
foreach(case ${TEST_CASES})
  add_test(NAME ${case} COMMAND test_spank_qrmi ${case})
endforeach()

# Known bugs on main. Each test passes once the bug is fixed; then drop it from this list.
# fresh_token_replaces_job_env_token: a token already in the job environment (e.g. from a
#   parent QPU job) is kept instead of the one just acquired, fixed by spank-plugins PR #204.
# unknown_resource_in_list: the job runs with the resources that were found, fixed by
#   spank-plugins PR #209.
set_tests_properties(
  unknown_resource_in_list
  fresh_token_replaces_job_env_token
  PROPERTIES WILL_FAIL TRUE)
